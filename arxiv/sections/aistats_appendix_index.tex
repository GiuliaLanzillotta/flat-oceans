\newcommand{\AppendixContentsRecord}[2]{%
  \ifnum\value{section}<3\def\apxcontentsfile{apxleft}%
  \else\def\apxcontentsfile{apxright}\fi
  \addtocontents{\apxcontentsfile}{\protect\contentsline{#1}%
    {\protect\numberline{\csname the#1\endcsname}#2}{\thepage}{\@currentHref}}}

\let\AppendixOriginalSubsubsection\subsubsection

\renewcommand{\subsubsection}[1]{\AppendixOriginalSubsubsection{#1}\AppendixContentsRecord{subsubsection}{#1}}
